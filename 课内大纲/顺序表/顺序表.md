12.已知两个长度分别为m和n的升序链表，若将它们合并为一个长度为m+n的降序链表，则最坏情况下的时间复杂度是多少？
$O(m+n)$

**注**：这里要用头插法
```cpp
struct ListNode {
    int val;
    ListNode * next;
    ListNode(int x): val(x), next(nullptr) {}
};

ListNode * mergeToDescending(ListNode * l1, ListNode * l2) {
    // 我们的结果链表头，初始为 NULL，它将作为“新楼”的基座
    ListNode * resultHead = nullptr;
    ListNode * p1 = l1;
    ListNode * p2 = l2;

    // 第一阶段：当两个链表都不为空时，进行“抢夺”
    while (p1 != nullptr && p2 != nullptr) {
        ListNode * target = nullptr;

        // 挑选较小的节点（为了让最终降序，必须先放小的，后放大的）
        if (p1 - > val <= p2 - > val) {
            target = p1;
            p1 = p1 - > next; // 物理指针向前移动，保护好剩余的链表
        } else {
            target = p2;
            p2 = p2 - > next;
        }

        // --- 核心：头插法逻辑 ---
        target - > next = resultHead; // 让⭐新节点的 next 指向当前“楼顶”
        resultHead = target; // 更新“楼顶”位置为新插入的节点
    }

    // 第二阶段：处理剩余的节点（注意！这里依然要用头插，否则会变成局部升序）
    ListNode * remaining = (p1 != nullptr) ? p1 : p2;
    while (remaining != nullptr) {
        ListNode * target = remaining;
        remaining = remaining - > next;

        target - > next = resultHead;
        resultHead = target;
    }

    return resultHead;
}
```
