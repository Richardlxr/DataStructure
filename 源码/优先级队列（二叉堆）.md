#include <algorithm>

#include <cstddef>

#include <cstdint>

#include <stdexcept>

#include <vector>

template < typename T > class HardcorePriorityQueue {
    private: std::vector < T > data; // 物理底层：极其紧凑的连续数组

    // 内部方法：从底向上的渗透（你的战场）
    void shiftUp(size_t index) {
        // 1. 挖坑保护：把新来的 VIP 攥在 CPU 寄存器里（或者栈内存），原位置 index 变成一个物理空洞
        T vip = std::move(data[index]);

        // 2. 物理边界防守与向上挑战
        // 只要坑位还没有升到最顶端（index > 0），就继续挑战
        while (index > 0) {
            // 极速算出亲爹的坐标
            size_t father = (index - 1) >> 1;

            // 3. 实力对比
            // 如果亲爹比 VIP 强（或一样强），说明 VIP 已经到达了它的天花板，立刻终止挑战！
            if (data[father] >= vip) {
                break;
            }

            // 4. 亲爹坠落，坑位上移（极其极致的单向物理覆盖，省去 swap 的 3 次读写！）
            // 亲爹守不住位置了，把他拽下来，填到当前的坑里
            data[index] = std::move(data[father]);

            // 坑位在物理逻辑上向上移动到了亲爹的位置
            index = father;
        }

        // 5. 尘埃落定：当循环结束，index 就是 VIP 最终赢下的王座，把 VIP 稳稳地放进去
        data[index] = std::move(vip);
    }
    T pop() {
        if (data.empty()) {
            throw std::out_of_range("物理抽干，堆已空！");
        }

        // 1. 恭送大佬：把 0 号王座的 VIP 攥在手里，准备最后 return
        T vip = std::move(data[0]);

        // 2. 揪出屌丝：把数组最后一个坑位的人拔出来，当作我们的下沉兵（low）
        T low = std::move(data.back());

        // 3. 物理切除：极其冷酷地销毁最后一个坑位，维护完全二叉树的物理紧凑性
        data.pop_back();

        // 防守：如果刚弹出的就是最后一个人，那系统就空了，直接结束
        if (!data.empty()) {
            size_t index = 0; // 坑位初始在王座
            size_t size = data.size();

            // 4. 边界防守：只要左儿子还在物理数组的边界内（说明还没跌到底层），就继续循环
            while ((index << 1) + 1 < size) {
                size_t left = (index << 1) + 1;
                // 物理连贯性：右儿子永远紧挨着左儿子
                size_t right = left + 1;

                // 5. 权力角逐：默认先假设左儿子最强
                size_t largest = left;

                // 如果右儿子存在，且右儿子比左儿子更强，最强者易主
                if (right < size && data[right] > data[left]) {
                    largest = right;
                }

                // 6. 阶级跨越检验：如果屌丝比最强的儿子还要强（或平起平坐）
                // 说明他镇得住现在的场子了，立刻终止坠落！
                if (low >= data[largest]) {
                    break;
                }

                // 7. 强者上位，坑位下移：让最强的儿子上位，原来的坑位跌落到儿子的位置
                data[index] = std::move(data[largest]);
                index = largest;
            }

            // 8. 尘埃落定：把屌丝稳稳地砸进他最终该待的物理坑位
            data[index] = std::move(low);
        }

        return vip; // 将大佬交还给外部调用者
    }

    public: HardcorePriorityQueue() {}

    // O(log N) 的极速入队n
    void push(const T & value) {
        data.push_back(value); // 蛮力塞入尾部（挖出一个初始坑位）
        shiftUp(data.size() - 1); // 启动重力引擎，让他自己浮上去
    }

    // ... 省略 pop 和 shiftDown
};
