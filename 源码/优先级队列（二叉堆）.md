#include <cstddef>

#include <stdexcept>

template < class T > class priorityQueue {
    private: int capacity; // 堆最多能容纳的元素个数
    int size; // 当前堆中的元素个数
    T * data; // 实际分配 capacity+1 个槽，data[0] 闲置，data[1..size] 有效

    public:
        // 构造函数
        // cap：初始容量（默认 8，即最多存 8 个元素）
        priorityQueue(std::size_t cap = 8): capacity(cap),
    size(0) {
        data = new T[capacity + 1]; // +1：下标 0 闲置，换来 1-based 索引更简洁的父子运算
    }

    // 扩容：容量翻倍
    void doubleSpace() {
        T * newdata = new T[(capacity << 1) + 1];
        for (int i = 1; i <= size; i++) { // 只复制有效元素 data[1..size]
            newdata[i] = data[i];
        }
        delete[] data;
        data = newdata;
        capacity = capacity << 1;
    }

    // 入队：把 _data 插入堆，并向上过滤（上浮）维持堆性质
    void enqueue(T _data) {
        if (size == capacity) {
            doubleSpace();
        }
        int hole = ++size; // 先在末尾开一个空穴

        // hole > 1：显式判断还没到根节点
        // _data < data[hole >> 1]：新元素比父节点小，需要继续上浮
        while (hole > 1 && _data < data[hole >> 1]) {
            data[hole] = data[hole >> 1]; // 父节点下移填空穴
            hole = hole >> 1; // 空穴上移到父节点位置
        }
        data[hole] = _data; // 找到了合适的位置，填入
    }

    /**
     * ╔══════════════════════════════════════════════════════╗
     * ║          建堆函数（Floyd 线性建堆法）                  ║
     * ╚══════════════════════════════════════════════════════╝
     *
     * 【功能】
     *   把一个普通的无序数组，一次性调整成一棵合法的最小堆。
     *
     * 【两种建堆方式的对比】
     *   方式一：逐个 enqueue()  → O(n log n)  慢
     *   方式二：Floyd 建堆法    → O(n)        快  ← 本函数采用
     *
     * 【核心思想：数组 = 完全二叉树】
     *   用数组下标来模拟父子关系，下标从 1 开始：
     *
     *              data[1]              ← 根节点（堆顶，最小值）
     *             /        \
     *         data[2]     data[3]
     *         /    \       /    \
     *     data[4] data[5] data[6] data[7]
     *
     *   规律：
     *     节点 i 的左孩子  = 2*i
     *     节点 i 的右孩子  = 2*i + 1
     *     节点 i 的父节点  = i / 2
     *
     * 【为什么从 size/2 开始？】
     *   下标 > size/2 的节点全是叶子节点（没有孩子），
     *   单个叶子天然满足堆性质，不用处理。
     *   只需从最后一个"有孩子的节点"（下标 = size/2）
     *   向左、向上逐个执行"向下过滤"，就能建出整棵堆。
     *
     * 【时间复杂度】O(n)
     * 【空间复杂度】O(1)（不算存储数组本身）
     *
     * @param arr  原始无序数组（下标从 0 开始的普通 C 数组）
     * @param n    数组中元素的个数
     */
    void buildHeap(T * arr, int n) {
        // ── 第一步：按需扩容 ─────────────────────────────────────────
        if (n > capacity) {
            delete[] data;
            capacity = n;
            data = new T[capacity + 1]; // +1：下标 0 闲置
        }

        // ── 第二步：把外部数组的数据拷入堆（堆下标从 1 开始）──────────
        size = n;
        for (int i = 0; i < n; i++) {
            data[i + 1] = arr[i]; // arr[0]→data[1], arr[1]→data[2], …
        }

        // ── 第三步：Floyd 建堆 ────────────────────────────────────────
        //
        // 从最后一个非叶节点（下标 = size/2）开始，一路往根的方向走，
        // 对每个节点调用 percolateDown()，让它"沉"到正确的位置。
        //
        // 以 7 个元素为例（size/2 = 3），处理顺序：
        //   i=3 → i=2 → i=1（根）
        //
        for (int i = size / 2; i >= 1; i--) {
            percolateDown(i);
        }
        // 循环结束：每个父节点都 ≤ 它的两个子节点，堆性质成立！
    }

    // 出队：返回并删除堆顶最小元素
    T dequeue() {
        if (size == 0) {
            throw std::out_of_range("队列空了，没东西可出了！");
        }

        T minItem = data[1]; // 1. 保存堆顶（最小值）
        T lastItem = data[size]; // 2. 取出最后一个元素
        size--; // 3. 逻辑上删掉最后一个位置

        if (size > 0) {
            data[1] = lastItem; // 4. 把最后一个元素填入堆顶的"坑"
            percolateDown(1); // 5. 让它向下沉，恢复堆性质
        }

        return minItem; // 6. 返回最小值
    }

    private:
        // 向下过滤（核心辅助函数）
        // 作用：让 data[hole] 不断与较小的孩子比较，沉到正确位置
        // hole：起始空穴下标
        void percolateDown(int hole) {
            int child;
            T tmp = data[hole]; // 暂存要下沉的元素

            for (; hole * 2 <= size; hole = child) {
                child = hole * 2; // 先指向左孩子

                // 选出左右孩子中较小的那个
                if (child != size && data[child + 1] < data[child]) {
                    child++; // 右孩子更小，改选右孩子
                }

                // 如果较小的孩子比 tmp 还小，孩子上移（tmp 继续下沉）
                if (data[child] < tmp) {
                    data[hole] = data[child];
                } else {
                    break; // tmp 比两个孩子都小，找到了正确位置
                }
            }
            data[hole] = tmp; // 把 tmp 放入最终位置
        }
};
